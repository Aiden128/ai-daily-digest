     7|## Linux Kernel 深度解析 — CPU Scheduling：CFS 從 `schedule()` 到 `pick_next_entity` 的原始碼全徑
     8|
     9|### 原理概述
    10|
    11|Linux 核心自 2.6.23 引入 Completely Fair Scheduler（CFS）取代 O(1) 排程器，其核心設計理念是讓每個行程獲得「公平的 CPU 時間份額」。CFS 不再維護傳統的時間片陣列，而是以紅黑樹（Red-Black Tree）管理 `vruntime`（虛擬執行時間），每次都選擇樹中 `vruntime` 最小的實體執行，從而實現 O(log n) 的挑選複雜度與精確的公平性。這份解析將從 `schedule()` 入口開始，追蹤到 CFS 挑選下一個任務的完整原始碼路徑。
    12|

    15|### 核心資料結構與函數

    59|
    60|`task_struct` 是核心中最重要的資料結構之一，每個行程（或執行緒）都對應一個實例。排程相關的欄位集中在 `se`（CFS 排程實體）、`rt`（即時排程實體）、`dl`（截止期限排程實體）以及 `sched_class`（指向其所屬的排程類別）。`on_rq` 標示該行程是否位於某個執行佇列上，`prio` 與 `static_prio` 則決定其靜態與動態優先權。

    64|#### 2. `struct sched_entity` — CFS 排程實體
    65|**File path**: `include/linux/sched.h`（第 575 行起）
    66|
    67|```c
    68|struct sched_entity {
    69|    /* For load-balancing: */
    70|    struct load_weight        load;
    71|    struct rb_node            run_node;
    72|    u64                       deadline;
    73|    u64                       min_vruntime;
    74|    u64                       min_slice;

    88|    s64                       vlag;
    89|    /* 'Protected' deadline, to give out minimum quantums: */
    90|    u64                       vprot;
    91|    u64                       slice;
    92|
    93|    u64                       nr_migrations;
    94|
    95|#ifdef CONFIG_FAIR_GROUP_SCHED
    96|    int                       depth;
    97|    struct sched_entity       *parent;
    98|    struct cfs_rq             *cfs_rq;
    99|    struct cfs_rq             *my_q;
   100|    unsigned long             runnable_weight;
   101|#endif
   102|
   103|    /* Per entity load average tracking. */
   104|    struct sched_avg          avg;
   105|};
   106|```
   107|
   108|`sched_entity` 是 CFS 的核心抽象。`vruntime` 記錄該實體已累積的虛擬執行時間，`run_node` 則是紅黑樹節點，用於將實體掛入 `cfs_rq->tasks_timeline`。`load.weight` 決定該實體的權重（由 nice 值換算而來），權重越高，相同實體時間內累積的 `vruntime` 增量越小，從而獲得更多實際 CPU 時間。`min_vruntime` 與 `vlag` 則用於進階的 lag-based 排程與喚醒延遲處理。

   117|    struct load_weight        load;
   118|    unsigned int              nr_queued;
   119|    unsigned int              h_nr_queued;
   120|    unsigned int              h_nr_runnable;
   121|    unsigned int              h_nr_idle;

   135|    struct sched_entity       *next;
   136|
   137|    /* CFS load tracking */
   138|    struct sched_avg          avg;

   154|    int                       idle;

   158|
   159|`cfs_rq` 代表一個 CFS 執行佇列，每顆 CPU 擁有一個根層級的 `cfs_rq`，並透過 `tasks_timeline`（型別為 `struct rb_root_cached`）維護所有可執行實體的紅黑樹。`nr_queued` 記錄樹中的實體數量，`curr` 指向目前正在執行的實體。當 `CONFIG_FAIR_GROUP_SCHED` 啟用時，`cfs_rq` 也支援群組排程（cgroups 的 cpu 子系統），此時每個 `task_group` 都可能擁有自己的 `cfs_rq`，形成階層式結構。

   164|**File path**: `kernel/sched/sched.h`（第 1131 行起）

   167|struct rq {
   168|    unsigned int              nr_running;

   173|    unsigned int              ttwu_pending;

   177|    struct task_struct __rcu  *curr;

   179|
   180|    u64                       nr_switches ____cacheline_aligned;

   184|    struct rt_rq              rt;
   185|    struct dl_rq              dl;
   186|
   187|    struct sched_dl_entity    fair_server;

   198|    unsigned int              clock_update_flags;

   202|
   203|`rq` 是每顆 CPU 的全域執行佇列，`nr_running` 記錄該 CPU 上所有排程類別的可執行任務總數。`cfs`、`rt`、`dl` 分別對應 CFS、RT（即時）、DL（Deadline）子佇列。`rq->curr` 指向當前真正在 CPU 上執行的 `task_struct`，而 `rq->donor` 則是在 `CONFIG_SCHED_PROXY_EXEC` 啟用時代表「排程器上下文」的捐贈者任務（proxy-exec 機制）。`nr_switches` 統計上下文切換次數，是 `vmstat` 中 `ctxt` 計數的來源之一。

   224|### 源碼流程說明
   225|
   226|#### 1. `schedule()` — 主動放棄 CPU 的入口
   227|
   228|```c
   229|asmlinkage __visible void __sched schedule(void)
   230|{
   231|    struct task_struct *tsk = current;
   232|
   233|    if (!task_is_running(tsk))
   234|        sched_submit_work(tsk);
   235|    __schedule_loop(SM_NONE);
   236|    sched_update_worker(tsk);
   237|}
   238|EXPORT_SYMBOL(schedule);
   239|```
   240|
   241|當行程呼叫 `schedule()`（例如執行 `sched_yield()`、進入可中斷睡眠、或系統呼叫返回前檢查到 `TIF_NEED_RESCHED` 標記），核心首先透過 `sched_submit_work()` 處理行程可能持有的鎖定（`__schedule_loop(SM_NONE)` 會不斷重複呼叫 `__schedule()` 直到 `need_resched()` 為假）。
   242|
   243|#### 2. `__schedule()` — 排程心臟
   244|
   245|```c
   246|static void __sched notrace __schedule(int sched_mode)
   247|{
   248|    struct task_struct *prev, *next;
   249|    bool preempt = sched_mode > SM_NONE;
   250|    struct rq_flags rf;
   251|    struct rq *rq;
   252|    int cpu;
   253|
   254|    cpu = smp_processor_id();
   255|    rq = cpu_rq(cpu);
   256|    prev = rq->curr;
   257|
   258|    local_irq_disable();
   259|    rcu_note_context_switch(preempt);
   260|    /* ... */
   261|    rq_lock(rq, &rf);
   262|    smp_mb__after_spinlock();
   263|
   264|    update_rq_clock(rq);
   265|
   266|    /* 若為自願排程且 prev 已進入睡眠狀態 */
   267|    if (!preempt && prev->__state) {
   268|        try_to_block_task(rq, prev, &prev_state, !task_is_blocked(prev));
   269|        switch_count = &prev->nvcsw;
   270|    }
   271|
   272|pick_again:
   273|    next = pick_next_task(rq, rq->donor, &rf);
   274|    rq->next_class = next->sched_class;
   275|    /* ... proxy-exec 處理 ... */
   276|
   277|picked:
   278|    clear_tsk_need_resched(prev);
   279|    clear_preempt_need_resched();
   280|
   281|    if (likely(prev != next)) {
   282|        rq->nr_switches++;
   283|        RCU_INIT_POINTER(rq->curr, next);
   284|        /* Also unlocks the rq: */
   285|        rq = context_switch(rq, prev, next, &rf);
   286|    } else {
   287|        rq_unpin_lock(rq, &rf);
   288|        __balance_callbacks(rq, NULL);
   289|        raw_spin_rq_unlock_irq(rq);
   290|    }
   291|}
   292|```
   293|
   294|`__schedule()` 的執行順序極為關鍵：
   295|1. **關閉本地中斷**（`local_irq_disable()`），防止在鎖定佇列期間發生中斷導致重入競爭。
   296|2. **取得 `rq->lock`**（`rq_lock(rq, &rf)`），這顆自旋鎖保護該 CPU 的執行佇列。
   297|3. **更新佇列時脈**（`update_rq_clock(rq)`），確保後續的時間統計基於正確的時間基準。
   298|4. **挑選下一個任務**（`pick_next_task(rq, rq->donor, &rf)`）。這是一個遍歷排程類別的函數：它依次嘗試 `stop_sched_class`、`dl_sched_class`、`rt_sched_class`、`fair_sched_class`、`idle_sched_class`，優先權由高至低。
   299|5. **清除重排程標記**（`clear_tsk_need_resched(prev)`）。
   300|6. **若 `prev != next`**，更新 `rq->curr` 並呼叫 `context_switch()` 執行架構相關的上下文切換（包含 `switch_mm()` 切換位址空間、`switch_to()` 切換暫存器狀態）。
   301|7. **若 `prev == next`**，直接解鎖返回，避免無意義的切換開銷。
   302|
   303|#### 3. `pick_next_task_fair()` — CFS 類別的挑選邏輯
   304|
   305|```c
   306|struct task_struct *
   307|pick_next_task_fair(struct rq *rq, struct task_struct *prev, struct rq_flags *rf)
   308|{
   309|    struct sched_entity *se;
   310|    struct task_struct *p;
   311|    int new_tasks;
   312|
   313|again:
   314|    p = pick_task_fair(rq, rf);
   315|    if (!p)
   316|        goto idle;
   317|    se = &p->se;
   318|
   319|#ifdef CONFIG_FAIR_GROUP_SCHED
   320|    if (prev->sched_class != &fair_sched_class)
   321|        goto simple;
   322|
   323|    /* 若 prev 與 p 不同，盡量減少 put/set 整個 cgroup 階層 */
   324|    if (prev != p) {
   325|        struct sched_entity *pse = &prev->se;
   326|        struct cfs_rq *cfs_rq;
   327|        while (!(cfs_rq = is_same_group(se, pse))) {
   328|            int se_depth = se->depth;
   329|            int pse_depth = pse->depth;
   330|            if (se_depth <= pse_depth) {
   331|                put_prev_entity(cfs_rq_of(pse), pse);
   332|                pse = parent_entity(pse);
   333|            }
   334|            if (se_depth >= pse_depth) {
   335|                set_next_entity(cfs_rq_of(se), se, true);
   336|                se = parent_entity(se);
   337|            }
   338|        }
   339|        put_prev_entity(cfs_rq, pse);
   340|        set_next_entity(cfs_rq, se, true);
   341|        __set_next_task_fair(rq, p, true);
   342|    }
   343|    return p;
   344|
   345|simple:
   346|#endif
   347|    put_prev_set_next_task(rq, prev, p);
   348|    return p;
   349|
   350|idle:
   351|    if (rf) {
   352|        new_tasks = sched_balance_newidle(rq, rf);
   353|        if (new_tasks < 0)
   354|            return RETRY_TASK;
   355|        if (new_tasks > 0)
   356|            goto again;
   357|    }
   358|    return NULL;
   359|}
   360|```
   361|
   362|`pick_next_task_fair()` 本身處理了群組排程的優化：當新選出的任務與前一個任務位於不同的 `task_group` 時，它只對需要變動的階層呼叫 `put_prev_entity()` 與 `set_next_entity()`，而非完整遍歷整棵樹。這在容器密集場景下顯著降低常數因子。
   363|
   364|#### 4. `pick_task_fair()` — 紅黑樹最小值挑選
   365|
   366|```c
   367|static struct task_struct *pick_task_fair(struct rq *rq, struct rq_flags *rf)
   368|{
   369|    struct sched_entity *se;
   370|    struct cfs_rq *cfs_rq;
   371|    struct task_struct *p;
   372|    bool throttled;
   373|
   374|again:
   375|    cfs_rq = &rq->cfs;
   376|    if (!cfs_rq->nr_queued)
   377|        return NULL;
   378|
   379|    throttled = false;
   380|    do {
   381|        if (cfs_rq->curr && cfs_rq->curr->on_rq)
   382|            update_curr(cfs_rq);
   383|
   384|        throttled |= check_cfs_rq_runtime(cfs_rq);
   385|        se = pick_next_entity(rq, cfs_rq, true);
   386|        if (!se)
   387|            goto again;
   388|        cfs_rq = group_cfs_rq(se);
   389|    } while (cfs_rq);
   390|
   391|    p = task_of(se);
   392|    if (unlikely(throttled))
   393|        task_throttle_setup_work(p);
   394|    return p;
   395|}
   396|```
   397|
   398|這是 CFS 挑選的核心迴圈：
   399|1. 取得該 CPU 的根 `cfs_rq`。
   400|2. 若 `cfs_rq->curr` 仍在佇列上，先呼叫 `update_curr()` 將目前任務的執行時間記帳到 `vruntime`（這一步至關重要，否則目前任務的 `vruntime` 會過小，導致它永遠被選中，形成「佔據 CPU」的假象）。
   401|3. 呼叫 `pick_next_entity()` 從紅黑樹左側（最小 `vruntime`）取出下一個實體。
   402|4. 若啟用群組排程且該實體是群組節點，則繼續向下深入該群組的 `cfs_rq`。
   403|
   404|#### 5. `update_curr()` — vruntime 的心臟
   405|
   406|```c
   407|static void update_curr(struct cfs_rq *cfs_rq)
   408|{
   409|    struct sched_entity *curr = cfs_rq->curr;
   410|    struct rq *rq = rq_of(cfs_rq);
   411|    s64 delta_exec;
   412|    bool resched;
   413|
   414|    if (unlikely(!curr))
   415|        return;
   416|
   417|    delta_exec = update_se(rq, curr);
   418|    if (unlikely(delta_exec <= 0))
   419|        return;
   420|
   421|    curr->vruntime += calc_delta_fair(delta_exec, curr);
   422|    resched = update_deadline(cfs_rq, curr);
   423|
   424|    if (entity_is_task(curr)) {
   425|        dl_server_update(&rq->fair_server, delta_exec);
   426|    }
   427|
   428|    account_cfs_rq_runtime(cfs_rq, delta_exec);
   429|
   430|    if (cfs_rq->nr_queued == 1)
   431|        return;
   432|
   433|    if (resched || !protect_slice(curr)) {
   434|        resched_curr_lazy(rq);
   435|        clear_buddies(cfs_rq, curr);
   436|    }
   437|}
   438|```
   439|
   440|`update_curr()` 的邏輯非常精簡但極其重要：
   441|- `delta_exec` 為自上次更新以來的實際執行時間（ns）。
   442|- `calc_delta_fair(delta_exec, curr)` 根據 `curr->load.weight` 對 `delta_exec` 進行縮放：權重越高的任務，其 `vruntime` 增量越小。這確保了高優先權任務能夠獲得更多實際 CPU 時間。
   443|- `update_deadline()` 更新 EEVDF（Earliest Eligible Virtual Deadline First）機制所需的 `deadline` 與 `slice`，這是 Linux 6.6+ 引入的替代傳統時間片的排程策略。
   444|- 若 `nr_queued > 1` 且目前實體的 slice 已用盡或不再受保護，則觸發 `resched_curr_lazy()` 設置 `TIF_NEED_RESCHED`，為搶佔鋪路。
   445|
   446|#### 6. `calc_delta_fair()` — 權重縮放公式
   447|
   448|```c
   449|static inline u64 calc_delta_fair(u64 delta, struct sched_entity *se)
   450|{
   451|    if (unlikely(se->load.weight != NICE_0_LOAD))
   452|        delta = __calc_delta(delta, NICE_0_LOAD, &se->load);
   453|
   454|    return delta;
   455|}
   456|```
   457|
   458|當 nice 值為 0（權重 `NICE_0_LOAD` = 1024）時，`vruntime` 增量等於實際執行時間。若 nice 值較低（高優先權，權重 > 1024），`__calc_delta()` 會將 `delta` 乘以 `NICE_0_LOAD / weight`，使得高優先權任務的 `vruntime` 增長更慢，從而更頻繁地被選中。
   459|
   460|#### 7. `sched_tick()` — 時脈中斷的排程脈動
   461|
   462|```c
   463|void sched_tick(void)
   464|{
   465|    int cpu = smp_processor_id();
   466|    struct rq *rq = cpu_rq(cpu);
   467|    struct task_struct *donor;
   468|    struct rq_flags rf;
   469|
   470|    sched_clock_tick();
   471|
   472|    rq_lock(rq, &rf);
   473|    donor = rq->donor;
   474|
   475|    psi_account_irqtime(rq, donor, NULL);
   476|    update_rq_clock(rq);
   477|
   478|    donor->sched_class->task_tick(rq, donor, 0);
   479|    calc_global_load_tick(rq);
   480|    sched_core_tick(rq);
   481|    scx_tick(rq);
   482|
   483|    rq_unlock(rq, &rf);
   484|    /* ... */
   485|}
   486|```
   487|
   488|每次時脈中斷（通常為 100~1000 Hz，取決於 `CONFIG_HZ`），`sched_tick()` 被呼叫。它更新 `rq` 時脈後，透過 `donor->sched_class->task_tick()` 呼叫對應排程類別的計時函數。對 CFS 而言，這會進入 `task_tick_fair()`，後者遍歷 `for_each_sched_entity()` 呼叫 `entity_tick()`，進而呼叫 `update_curr()` 與 `update_load_avg()`。若任務已用盡其時間配額，`update_curr()` 內的 `resched_curr_lazy()` 會設置重排程標記，讓該任務在下一次進入/返回使用者空間時被搶佔。

   492|### 設計原理與悲劇情境
   493|
   494|#### 1. EEVDF 取代純 vruntime 比較
   495|
   496|Linux 6.6 起，CFS 逐步導入 EEVDF（Earliest Eligible Virtual Deadline First）。傳統 CFS 單純比較 `vruntime` 最小值，這導致高 nice 值（低優先權）任務可能長時間無法獲得 CPU（「餓死」問題），因為它們的 `vruntime` 永遠追不上不斷被重置的高優先權任務。EEVDF 為每個實體賦予 `slice`（時間配額）與 `deadline = vruntime + slice`，只有當目前實體的 `deadline` 小於佇列中最小 `deadline` 時，才允許繼續執行；否則觸發搶佔。這確保了每個任務在有限時間內必定獲得執行機會。
