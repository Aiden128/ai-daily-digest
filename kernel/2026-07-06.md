# Linux Kernel 深度解析 — TCP/IP 核心子系統：`sk_buff`、Socket Layer 與 `tcp_v4_rcv` 的端到端原始碼全徑

## 原理概述

Linux TCP/IP 堆疊從使用者呼叫 `socket(AF_INET, SOCK_STREAM, 0)` 開始，經由 VFS 的 `sys_socket` 進入 `__sock_create`，在 `net_families[AF_INET]` 找到 `inet_create`，接著 `inet_init` 預先註冊好的 `tcp_prot` 把一個 `struct sock` 與 `struct socket` 綁定成 TCP 連線的兩端。資料進入 NIC 後，XDP/eBPF 提早 drop 或 redirect，否則走 `napi_gro_receive` → `netif_receive_skb` → `ip_rcv` → `ip_rcv_finish` → `ip_local_deliver` → `ip_local_deliver_finish` → `ipprot->handler`（對 TCP 來說是 `tcp_v4_rcv`）。`tcp_v4_rcv` 從 `__inet_lookup` 找到 `struct sock`、再把 skb 送進 `tcp_v4_do_rcv` → `tcp_rcv_state_process` 或 `tcp_v4_rcv` → `tcp_v4_do_rcv` → `tcp_data_queue` → `sock_def_readable`，使用者行程在 `tcp_recvmsg` 內透過 `skb_copy_datagram_iter` 取得封包。整條鏈路核心的資料載體是 `struct sk_buff`（暱稱 skb），它是一個帶有鏈結串列、frag_list、frags 與 GRO 狀態的可變長封包描述結構，每個 socket 用 `sk_buff_head`（`sk_receive_queue` / `sk_write_queue` / `sk_async_wait_queue`）與 skb 互動。

本文逐段解析這條從建立 socket、TX 路徑、到 RX 路徑、TCP 狀態機與 sk_buff 管理的所有重要源碼點。

## 核心資料結構

### `struct sk_buff` 與 `struct skb_shared_info`

路徑：`include/linux/skbuff.h`

```c
struct sk_buff {
    union {
        __u8            headers[0];
        struct {
            /* 對齊到 32 位元，先放高頻率存取的欄位 */
            struct sk_buff          *next;
            struct sk_buff          *prev;

            union {
                struct net_device   *dev;
                unsigned long       dev_scratch;
            };
        };
    };

    struct sock              *sk;            /* 所屬 socket；不可 NULL（控制緩衝除外） */

    /* 傳輸層與網路層的 offset；所有存取巨集都透過它們索引 */
    __u16                    transport_header;
    __u16                    network_header;
    __u16                    mac_header;

    __u16                    inner_transport_header;
    __u16                    inner_network_header;
    __u16                    inner_mac_header;

    sk_buff_data_t           tail;           /* skb->data + len，線性資料尾端 */
    sk_buff_data_t           end;           /* 線性 buffer 結尾（dev->alloc 範圍） */
    unsigned char            *head, *data;   /* head <= data <= tail <= end */

    __u32                    len;            /* 線性部分 + frag_list 總長度 */
    __u16                    mac_len;
    __u16                    hdr_len;
    __u16                    queue_mapping;  /* 對應 TX queue；XPS 用 */
    __u8                     vlan_present:1, inner_vlan_present:1;
    __u8                     tc_skip_classify:1;
    __u16                    gso_size;       /* GSO 段大小；0 表示不走 GSO */
    __u16                    gso_segs;
    struct skb_shared_hwtstamps hwtstamps;
    __be16                   protocol;       /* ETH_P_* 決定走 ipv4/ipv6/arp */
    __u16                    vlan_tci;
    __u8                     dev_priority:7;
    __u8                     local_df:1;     /* 設定 IP_DF */
    __u8                     pkt_type:3,     /* PACKET_HOST / BROADCAST / ... */
                                ignore_df:1,
                                mirrored:1,
                                ipvs_property:1,
                                inner_protocol_type:1,
                                network_offset:1,
                                __pkt_type_offset:1;
    __u8                     active_extensions;  /* SKB_EXT_* */

    struct skb_shared_info   *shinfo;        /* 與 skb->end 同一位置；線性資料末端就是它 */
    /* ... 略 ... */
};
```

`struct skb_shared_info` 緊接在 `skb->end` 之後，是 `skb_shinfo(SKB)` 巨集回傳的物件。重要欄位：

```c
struct skb_shared_info {
    __u8          __unused;
    __u8          meta_len;
    __u8          nr_frags;        /* frags[] 中已用項目數 */
    __u8          tx_flags;
    unsigned int  gso_size;
    unsigned int  gso_segs;
    struct sk_buff *frag_list;     /* 鏈結的子 skb（用於超大封包，jumbogram） */
    struct skb_shared_hwtstamps hwtstamps;
    unsigned int  gso_type;        /* SKB_GSO_TCPV4 / UDP / ... */
    u32           tskey;
    atomic_t      dataref;         /* 對 skb->head 引用計數 */
    void         *destructor_arg;
    /* frags[] 結尾；實際大小由 kmem_cache 決定 */
    union {
        struct bio_vec   frags[MAX_SKB_FRAGS];   /* x86 通常為 16 */
    };
};
```

**設計原理：** 把 metadata（`skb_shared_info`）緊接在線性資料尾端，能在 `skb->head` 一次 `kmalloc` 中分配所有東西，clone 時只要遞增 `dataref` 並複製 skb 本體，省一次 alloc 與一次 free（`skb_clone` 路徑）。`tcp_wmem` / `tcp_rmem` 透過 `skb_expand_head` 重新分配並保留 `frags[]`。

### `struct sock`（IPv4 TCP 為例）

路徑：`include/net/sock.h` 與 `include/linux/tcp.h`

`inet_sock` 與 `inet_connection_sock` 與 `tcp_sock` 形成三層繼承：

```c
struct inet_sock {
    struct sock               sk;
    __be32                   inet_saddr;        /* 源位址（小端序） */
    __be16                   inet_sport;        /* L4 源 port */
    __u16                    inet_id;           /* IP id，用於分片 */
    __u8                     inet_tos;          /* TOS/DSCP */
    struct ip_options_rcu    *inet_opt;          /* 來源路徑 IP options */
    int                      inet_sndbuf;        /* 對映到 sk_sndbuf */
    /* ... */
};

struct inet_connection_sock {
    struct inet_sock          icsk_inet;
    struct request_sock_queue icsk_accept_queue; /* 半連線 / 完整 SYN queue */
    struct timer_list         icsk_retransmit_timer;
    struct timer_list         icsk_delack_timer; /* 200ms 延遲 ACK */
    struct timer_list         icsk_pmtu_timer;   /* Path MTU 探測 */
    u32                       icsk_rto;          /* 重傳逾時 */
    /* ... */
};

struct tcp_sock {
    struct inet_connection_sock  inet_conn;
    __u32                        tcp_mstamp;       /* tcp_clock_cache 取樣 */
    u32                          rcv_nxt;          /* 期望收到的下一個 seq */
    u32                          snd_nxt;          /* 下一個要送的 seq */
    u32                          snd_una;          /* 已送出但未 ack 的最早 seq */
    u32                          snd_wl1, snd_wl2; /* 視窗更新時的 seq 用於 sack */
    u32                          copied_seq;       /* 應用層已讀到的 seq */
    u32                          rcv_wup;          /* 最近一次視窗更新點 */
    struct sk_buff_head          out_of_order_queue;
    struct sk_buff_head          tcp_tsorted_anchor;   /* 用於 RACK 偵測丟包 */
    /* ... SACK、擁塞控制 cstate、TSQ、reordering... */
};
```

**設計原理：** 將高頻欄位放 `sock_common`（第一個 cache line），如 `skc_daddr` / `skc_state` / `skc_bind_node`，讓 RX 路徑在 `__inet_lookup` 只觸碰 1 個 cache line 就完成 hash 對比；其餘低頻欄位往後放，減少 false sharing。

## Socket 建立路徑：`socket(2)` → `inet_create`

### 1. `sys_socket` 與 `__sock_create`

路徑：`net/socket.c`

```c
SYSCALL_DEFINE3(socket, int, family, int, type, int, protocol)
{
    return __sys_socket(family, type, protocol);
}

int __sys_socket(int family, int type, int protocol)
{
    struct socket *sock;
    int flags;

    /* SOCK_* <-> SOCK_CLOEXEC / SOCK_NONBLOCK 拆解 */
    flags = type & ~SOCK_TYPE_MASK;
    type  = type & SOCK_TYPE_MASK;

    if (SOCK_NONBLOCK != O_NONBLOCK && (flags & SOCK_NONBLOCK))
        flags = (flags & ~SOCK_NONBLOCK) | O_NONBLOCK;

    sock = sockfd_lookup_light(...);    /* 已被忽略；保留給舊介面 */

    if (likely(!sock)) {
        retval = __sock_create(family, type, protocol, &sock, flags);
        if (retval >= 0)
            retval = map_fd(retval, sock, flags);  /* 分配 fd 並回傳 */
    }
    return retval;
}

int __sock_create(int family, int type, int protocol,
                  struct socket **res, int kern)
{
    struct net *net = current->nsproxy->net_ns;
    const struct net_proto_family *pf;
    struct socket *sock;
    int err;

    /* 安全模組檢查；SELinux/AppArmor 攔在這 */
    err = security_socket_create(family, type, protocol, kern);
    if (err) return err;

    sock = sock_alloc();   /* 從 sock_inode_cache 取得 struct socket + inode */
    if (!sock) return -ENOMEM;

    sock->type    = type;
    sock->ops     = ERR_PTR(-ECONNREFUSED);   /* 防呆：未初始化前不得呼叫 */
    sock->sk      = NULL;                     /* sock->sk 必須由 protocol 設定 */

    pf = net_family_get(family);              /* net_families[family]，spin_lock */
    if (!pf) {
        err = -EAFNOSUPPORT;
        goto out_module_put;
    }

    err = pf->create(sock, protocol, kern);   /* → inet_create */
    if (err) goto out_module_put;

    (*res) = sock;
    return 0;
    /* ... */
}
EXPORT_SYMBOL(__sock_create);
```

### 2. `inet_create` 與 TCP 選擇

路徑：`net/ipv4/af_inet.c`

```c
static int inet_create(struct net *net, struct socket *sock, int protocol,
                       int kern)
{
    struct sock *sk;
    struct inet_protosw *answer;
    struct inet_sock *inet;
    /* type 對映 SOCK_STREAM -> IPPROTO_TCP（若 protocol 為 0） */
    static const struct inet_protosw protosw[] = {
        /* ... SOCK_STREAM, IPPROTO_TCP -> tcp_prot ... */
    };

    /* 看 protocol 是 IPPROTO_TCP / UDP / ICMP，掃過 inetsw[] 找到 protosw */
    list_for_each_entry_rcu(answer, &inetsw[sk->sk_prot->no_check ? 1 : 0], list) {
        /* 略 */
    }

    /* 分配 sk；大小依 prot->obj_size 決定 (TCP 為 sizeof(struct tcp_sock)) */
    sk = sk_alloc(net, PF_INET, GFP_KERNEL, answer->prot, kern);
    if (!sk) goto out;

    sock_init_data(sock, sk);   /* 將 sock->sk 雙向連結；初始化 skb 鏈結 */

    sk->sk_protocol = protocol; /* IPPROTO_TCP */

    /* 從 protosw 拷貝 prot 與 ops */
    sk->sk_prot     = answer->prot;
    sock->ops       = answer->ops;     /* inet_stream_ops */

    /* ... 設定 sk_policy/rcvbuf/sndbuf 預設值 ... */

    if (sk->sk_prot->init)
        sk->sk_prot->init(sk);     /* tcp_v4_init_sock */

    /* 將 sk 連結到 hash 槽或無；這時尚未 connect */
    return 0;
}
```

`answer->prot` 為 `tcp_prot`（位於 `net/ipv4/tcp_ipv4.c`），裡面定義了 `tcp_v4_init_sock` / `tcp_v4_connect` / `tcp_v4_rcv` / `tcp_v4_send_check` 等所有 TCP L4 callback。

## TCP TX 路徑：`tcp_sendmsg` → `tcp_push`

### `tcp_sendmsg` 主要步驟

路徑：`net/ipv4/tcp.c`

```c
int tcp_sendmsg(struct sock *sk, struct msghdr *msg, size_t size)
{
    struct inet_connection_sock *icsk = inet_csk(sk);
    struct tcp_sock *tp = tcp_sk(sk);
    struct sk_buff *skb;
    struct tcp_skb_cb *tcb;
    int flags, err, copied = 0;
    int mss_now = 0, size_goal, copied_syn = 0;
    bool process_backlog, zcopy = false;
    long timeo;

    lock_sock(sk);  /* BH-disabled；禁止在 softirq 上下文呼叫 */
    flags = msg->msg_flags;

    timeo = sock_sndtimeo(sk, flags & MSG_DONTWAIT);
    if (... sk_in_state(sk, TCP_LISTEN) ...)
        goto out_err;

    /* Force MSG_SPLICE_PAGES 若 socket 支援 */
    if (skb_appendable(sk, ...) || (sk_page_frag_refill(sk, ...) == -ENOMEM))
        return -ENOMEM;

    while (msg_data_left(msg)) {
        /* 1) 取得 mss_now 與 size_goal (含 TSO/GSO 計算) */
        if (copied)
            tcp_rate_check_app_limited(sk);
        mss_now = tcp_send_mss(sk, &size_goal, flags, &copied_syn);
        /* size_goal = mss_now + MAX_TCP_OPTION_SPACE = 16 或更多 */

        /* 2) 若 sk->sk_write_queue 末端 skb 還有空間，就 append */
        skb = tcp_write_queue_tail(sk);
        if (skb && can_coalesce(skb)) {
            /* 將 user data 直接拷貝到 skb 的 frags 線性區 */
            err = skb_append_data(skb, ...) ? : skb_copy_to_page_nocache();
        }

        /* 3) 否則配置新 skb 放入 sk_write_queue */
        if (!skb) {
            skb = sk_stream_alloc_skb(sk, 0, sk->sk_allocation, ...);
            if (!skb) ...;

            skb->ip_summed = CHECKSUM_PARTIAL;
            skb_entail(sk, skb);   /* tp->write_seq 寫入 tcb->seq；放入尾端 */
            tcb = TCP_SKB_CB(skb);
            tcb->end_seq   = tp->write_seq;
            tcb->tcp_flags = TCP_PSH | (copied_syn ? TCP_SYN : 0);
        }

        /* 4) 拷貝 user data 進入 skb */
        if (memcpy_from_msg(skb_put(skb, copy), msg, copy))
            goto do_fault;

        /* 5) 滿了就把 skb 推出去；不滿則繼續 fill */
        if (forced_push(tp)) {
            tcp_mark_push(tp, skb);
            tcp_push(sk, flags, mss_now, tp->nonagle, size_goal);
        }
    }
    /* 結束後再 tcp_push 一次 */
    tcp_push(sk, flags, mss_now, tp->nonagle, size_goal);
out:
    release_sock(sk);
    return copied;
}
```

### `tcp_push` 與 `__tcp_push_pending_frames`

```c
void tcp_push(struct sock *sk, int flags, int mss_now, int nonagle, int size_goal)
{
    struct tcp_sock *tp = tcp_sk(sk);
    struct sk_buff *skb;

    skb = tcp_write_queue_tail(sk);
    if (!skb) return;

    if (forced_push(tp) || tcp_skb_is_last(sk, skb))
        tcp_mark_push(tp, skb);

    tcp_mark_urg(tp, flags, skb);
    __tcp_push_pending_frames(sk, mss_now, nonagle);
}

void __tcp_push_pending_frames(struct sock *sk, unsigned int cur_mss,
                               int nonagle)
{
    struct sk_buff *skb = tcp_write_queue_tail(sk);
    struct tcp_sock *tp = tcp_sk(sk);

    if (unlikely(!skb)) return;

    if (!tcp_skb_pcount(skb) || tcp_skb_pcount(skb) == 1) {
        /* 沒有 GSO；直接發送 */
        tcp_chrono_start(sk, TCP_CHRONO_BUSY);
        tcp_transmit_skb(sk, skb, 0, (__force int)tcp_sk(sk)->sk_gso_disabled);
        return;
    }
    /* GSO 路徑：tcp_write_xmit / tcp_tso_set_state */
    tcp_write_xmit(sk, cur_mss, nonagle, 0, sk_gso_max_segs(sk), GFP_ATOMIC);
}
```

### `tcp_transmit_skb` — 真正進入 L3 之前

```c
int tcp_transmit_skb(struct sock *sk, struct sk_buff *skb, int clone_it,
                     int gfp_mask)
{
    const struct inet_connection_sock *icsk = inet_csk(sk);
    struct inet_sock *inet;
    struct tcp_sock *tp;
    struct tcp_skb_cb *tcb;
    struct tcphdr *th;
    int err;

    /* BUG_ON(sk_state == TCP_TIME_WAIT); */

    if (clone_it) {
        /* 1 個 skb 可能被多個 process 用；clone 為 cheap copy（只 ++dataref） */
        skb = skb_clone(skb, gfp_mask);
        if (unlikely(!skb))
            return -ENOBUFS;
        /* cloned 標記避免 free */
    }

    inet = inet_sk(sk);
    tp   = tcp_sk(sk);
    tcb  = TCP_SKB_CB(skb);
    memset(tcb, 0, sizeof(*tcb));

    /* 設定 skb->transport_header 與 seq/snd_nxt */
    skb_push(skb, tcp_header_size);
    skb_reset_transport_header(skb);
    th = (struct tcphdr *)skb->data;
    th->source        = inet->inet_sport;
    th->dest          = inet->inet_dport;
    th->seq           = htonl(tcb->seq);
    th->ack_seq       = htonl(tp->rcv_nxt);
    th->doff          = (tcp_header_size >> 2);
    th->window        = htons(tcp_select_window(sk));
    tcp_options_write((__be32 *)(th + 1), tp, &opts);
    th->urg_ptr       = htons(opts.flags & (TSOPT_HUMAN | TSOPT_HUMAN2) ?
                              tp->snd_up : 0);
    th->urg           = (opts.flags & TSOPT_HUMAN_URG) ? 1 : 0;
    th->check         = 0;
    th->fin           = (tcb->tcp_flags & TCP_FIN) ? 1 : 0;
    th->syn           = (tcb->tcp_flags & TCP_SYN) ? 1 : 0;
    th->rst           = (tcb->tcp_flags & TCP_RST) ? 1 : 0;
    th->psh           = (tcb->tcp_flags & TCP_PSH) ? 1 : 0;
    th->ack           = (tcb->tcp_flags & TCP_ACK) ? 1 : 0;

    if (likely(!(tcb->tcp_flags & TCPHDR_SYN)) && icsk->icsk_ca_state == TCP_CA_Open)
        tcp_event_ack_scheduled_check(sk, skb);

    /* ECN、CWR 旗標 */
    inet_csk(sk)->icsk_af_ops->send_check(sk, skb);   /* tcp_v4_send_check */

    /* 最後呼叫 IP 層 */
    err = ip_queue_xmit(sk, skb, ...);
    return err;
}
```

## 接收路徑：NIC → `tcp_v4_rcv` → `tcp_data_queue`

### 1. `ip_rcv` 入口

路徑：`net/ipv4/ip_input.c`

```c
int ip_rcv(struct sk_buff *skb, struct net_device *dev,
           struct packet_type *pt, struct net_device *orig_dev)
{
    struct net *net = dev_net(dev);

    skb = ip_rcv_core(skb, net);    /* 校驗 header、檢查長度、做路由表 prep */
    if (!skb) return NET_RX_DROP;

    return NF_HOOK(NFPROTO_IPV4, NF_INET_PRE_ROUTING,
                   net, NULL, skb, dev, NULL, ip_rcv_finish);
}

static int ip_rcv_finish(struct net *net, struct sock *sk, struct sk_buff *skb)
{
    struct net_device *dev = skb->dev;
    int ret;

    /* 確認 dst entry（RT lookup 已透過 NF_HOOK 之後做） */
    if (!skb_valid_dst(skb))
        ip_route_input_noref(skb, ip_hdr(skb)->saddr, ip_hdr(skb)->daddr,
                             ip_hdr(skb)->tos, dev);

    /* 跑 IP defrag 與 NF_INET_LOCAL_IN hook，最後 dst_input(skb) → ip_local_deliver */
    ret = dst_input(skb);
    return ret;
}
```

```c
int ip_local_deliver(struct sk_buff *skb)

static int ip_local_deliver_finish(struct net *net, struct sock *sk,
                                   struct sk_buff *skb)
{
    struct iphdr *iph = ip_hdr(skb);
    int protocol = iph->protocol;
    const struct net_protocol *ipprot;
    int raw = 0;

    /* 分片重組：ip_defrag() 會找對應的 INET frags bucket 並合併 */
    if (ip_is_fragment(iph)) {
        if (ip_defrag(net, skb, IP_DEFRAG_LOCAL_DELIVER))
            return 0;   /* 已被合併 / 排隊 */

    } else {
        raw = raw_local_deliver(skb, protocol);   /* 兜底：raw socket 收到 */
    }
    return ret;
}
```

### 3. `tcp_v4_rcv` 主流程

路徑：`net/ipv4/tcp_ipv4.c`

```c
int tcp_v4_rcv(struct sk_buff *skb)
{
    struct net *net;
    struct sock *sk;
    struct sk_buff *skb_to_free;
    int sdif = inet_sdif(skb);
    int dif = inet_iif(skb);
    __u32 seq, end_seq;
    bool thdr_ok;
    int drop_reason;

    if (skb->pkt_type == PACKET_HOST)
        ;   /* 走主流程 */
    else
        goto discard_it;

    if (skb->skb_iif)
        ;   /* 略 */

    /* 確認 IP header 與 IP options 已就緒；再做 L4 parse */
    if (!pskb_may_pull(skb, sizeof(struct tcphdr)))
        goto discard_it;

    th = (struct tcphdr *)skb->data;
    if (unlikely(th->doff < sizeof(struct tcphdr) / 4))
        goto bad_packet;

    if (!pskb_may_pull(skb, th->doff * 4))
        goto discard_it;

    /* GRO 過程中已經算過一次，這裡只做最終校驗 */
    tcp_checksum_done(skb);
    if (!skb_csum_unnecessary(skb) && tcp_v4_checksum_init(skb))
        goto csum_error;

    th = (struct tcphdr *)skb->data;
    skb->transport_header += th->doff * 4;
    TCP_SKB_CB(skb)->seq = ntohl(th->seq);
    TCP_SKB_CB(skb)->end_seq = TCP_SKB_CB(skb)->seq + th->syn + th->fin +
        !!(tcp_flag_word(th) & TCP_PSH) + skb->len - th->doff * 4;
    TCP_SKB_CB(skb)->ack_seq = ntohl(th->ack_seq);
    TCP_SKB_CB(skb)->tcp_flags = tcp_flag_word(th);
    TCP_SKB_CB(skb)->sacked = 0;
    TCP_SKB_CB(skb)->ip_dsfield = ipv4_get_dsfield(iph);

    /* 1) 找 socket：__inet_lookup_skb */
    sk = __inet_lookup_skb(net, skb, sdif, &refcounted);
    if (!sk)
        goto no_tcp_socket;

process:
    if (sk->sk_state == TCP_TIME_WAIT)
        goto do_time_wait;
    if (sk->sk_state == TCP_NEW_SYN_RECV) {
        /* special: syncookies/synproxy */
        sk = tcp_v4_cookie_check(skb, &iph);
        if (!sk)
            goto no_tcp_socket;
        /* 重新跑 lookup */
    }

    th = (struct tcphdr *)skb->data;
    skb->transport_header += th->doff * 4;

    skb_inward_action(skb, sk);  /* 追蹤 */
    skb->dev = NULL;

    bh_lock_sock_nested(sk);
    /* TIME_WAIT / LISTEN 走對應 fast path */
    tcp_v4_do_rcv(sk, skb);
    bh_unlock_sock(sk);

    /* ... */
}
EXPORT_SYMBOL(tcp_v4_rcv);
```

### 4. `__inet_lookup_skb` — 雜湊查表

路徑：`net/ipv4/inet_hashtables.c`

```c
struct sock *__inet_lookup_skb(struct net *net, struct sk_buff *skb,
                               int sdif, bool *refcounted)
{
    struct iphdr *iph = ip_hdr(skb);
    struct tcphdr *th;
    struct sock *sk;

                        iph->daddr, th->dest, sdif, iif, true);
    return sk;
}

static inline struct sock *__inet_lookup(struct net *net,
                                          struct inet_hashinfo *hashinfo,
                                          __be32 saddr, __be16 sport,
                                          __be32 daddr, __be16 dport,
                                          int sdif, int iif, bool log)
{

    struct inet_ehash_bucket *head = inet_ehash_bucket(hashinfo, hash);
    struct sock *sk;

    sk = inet_lookup_reusable_row(net, hashinfo, hash, ...);  /* reuseport */
    if (sk)

    /* 走 inet_lhash2 / bhash / ehash ... */

```

**設計原理：** `tcp_hashinfo` 在 `tcp_init` 中分配 4 個 hash 區（ehash / bhash / listening_hash / request_sock_hash），ehash 與 bhash 各自有自己的 spin lock。`inet_ehashfn` 用 `jhash_3words` 把四元組（src_ip, src_port, dst_ip, dst_port）壓成 hash，從而把 `O(n)` socket 表降到 `O(1)` 查表。

```c
{

    struct request_sock *req;
    struct tcp_sock *tp;
    enum skb_drop_reason reason;

    if (sk->sk_state == TCP_LISTEN) {

        if (sk == NULL) goto discard;

    ret = tcp_rcv_state_process(sk, skb);  /* 處理 state machine */
    if (ret) {
        if (sk_state_load(sk) == TCP_CLOSE)
            goto slow_path_no_error;   /* socket 被關閉，ret = 0 */
        ...
    }
    return 0;
}
```

```c
int tcp_rcv_state_process(struct sock *sk, struct sk_buff *skb)
{

    case TCP_FIN_WAIT2:

            tcp_data_queue(sk, skb);
            queued = 1;
            break;
        case TCP_FIN_WAIT1:
        case TCP_FIN_WAIT2:
            tcp_data_queue(sk, skb);
            ...
        case TCP_CLOSE_WAIT:
            tcp_data_queue(sk, skb);
            queued = 1;
            break;
        /* ... CLOSING / LAST_ACK / TIME_WAIT 等 */
        }
        break;
    }
    return 0;
}
```

```c
static void tcp_data_queue(struct sock *sk, struct sk_buff *skb)
{
    struct tcp_sock *tp = tcp_sk(sk);

            ...
        }

    } else {
        /* out-of-order：先看是否能填補 gap */
        tcp_data_queue_ofo(sk, skb);
```

**設計原理：** `tcp_data_queue_ofo` 將亂序包放入 `out_of_order_queue`（紅黑樹，以 seq 排序），這對 RACK loss detection 至關重要；當新的資料封包「填補」了 gap，RACK 核心 `tcp_rack_advance` 就能識別哪些重傳是必須的。

### 8. `skb_clone` 與 `__kfree_skb` — skb 生命週期

```c
struct sk_buff *skb_clone(struct sk_buff *skb, gfp_t gfp_mask)
{
    struct sk_buff *n;

    n = skb + 1;   /* 緊接在 skb 後面 */
    if (skb_orphan_frags(skb, gfp_mask))
        return NULL;

    /* header + tail 一起拷貝（不含線性資料） */
    *n = *skb;
    /* 重新鏈入 sock 自己的 skb 鏈（next/prev 在鏈結呼叫端做） */
    n->cloned = 1;
    n->fclone = SKB_FCLONE_UNAVAILABLE;
    n->destructor = NULL;
    skb->cloned = 1;

    /* 線性資料的引用計數 +1 */
    refcount_inc(&skb_shinfo(skb)->dataref);
    return n;
}
```

skb 釋放由 `__kfree_skb` 串接：

```c
void __kfree_skb(struct sk_buff *skb)
{
    /* 1) 若 skb->sk 仍繫結，detach */
    if (likely(atomic_read(&skb->users) == 1))
        smp_rmb();
    else if (likely(!atomic_dec_and_test(&skb->users)))
        return;
    /* 釋放 frags / destructor / skb->head */
    skb_release_all(skb);
    kfree_skbmem(skb);
}
EXPORT_SYMBOL(__kfree_skb);
```

**設計原理：** `skb->users` 為 1 時代表唯一持有者，可直接 `kfree`；被 `skb_clone` 後使用者 ≥2，最後一個釋放者會 `kfree_skbmem`。`skb_release_all` 處理 `skb->destructor`（如 TCP TX 排隊的 sock_efree）、`skb_release_data`（逐 frags put_page）、以及 `skb_release_head`。

```c

                                struct iov_iter *to, int len, bool short_ok,

{
    int end = skb_headlen(skb);     /* 線性部分長度 */

        if (copy_page_to_iter_nofail(virt_to_page(skb->data),
                                      offset_in_page(skb->data + offset),
                                      min(end - offset, len), to))
            return -EFAULT;
    }
    /* 再 iterate skb frags[] */
    for (i = -1; ...) {
        if (skb_copy_datagram_from_iter(skb, off, &to, copy) < 0)
            return -EFAULT;
    }
    return 0;
}
```

**設計原理：** `iov_iter` 抽象化 iovec、pipe、kvec，使用者空間讀取路徑不需 copy 進 core buffer；`skb->data` 線性部分 + `frags[]` 一起線性化到 user；當應用一次 `recv()` 收不完整個 skb 時，會保留 skb 本身以備下次再讀（這是 `skb_peek` 比 `skb_dequeue` 慢的原因之一）。

## 計時器、擁塞控制與 GRO/TSO 銜接

### 1. 計時器鏈

`tcp_init_xmit_timers(sk)` 註冊 4 個 timer：
- `icsk_retransmit_timer` (`tcp_retransmit_timer`)
- `icsk_delack_timer` (`tcp_delack_timer`)
- `sk_timer` (`tcp_keepalive_timer`，設定 SO_KEEPALIVE)
- `icsk_pmtu_timer` (`tcp_pmtu_discovery_timer`)

它們都由 `tcp_set_timeout` / `tcp_reset_xmit_timer` 控制啟動。

### 2. GRO → TSO 對稱

- **RX 路徑 GRO**：`tcp4_gro_receive` 將多個 TCP 段合併成一個 skb，減少上層處理成本。`skb->gro_*` 欄位由 GRO 追蹤。
- **TX 路徑 TSO**：`tcp_write_xmit` 將 `skb->len > mss_now` 的 skb 切成多個 MSS，由 `tcp_set_skb_tso_segs` 計算 `gso_segs`，網卡做硬體分割。
- **GRO 對 TSO 的鏡像：** 同一個 4-tuple 進 NIC 時，HSR 對等端重組後，skb 結構可一路不變返回 L4，這是 datacenter 提升 throughput 的關鍵。

### 3. cgroup / eBPF 鉤子

- `tcp_bpf_recvmsg` 與 `tcp_bpf_sendmsg` 為 `BPF_PROG_TYPE_SOCK_OPS` 提供 hook。
- `sk_filter_run` 在 `tcp_v4_rcv` 透過 `sk_filter` 跑 `BPF_PROG_TYPE_SOCKET_FILTER`。
- `cgroup_sock` 與 `cgroup_bpf` 共同控制 cgroup v2 的網路過濾。

## 設計原理與悲悸情境

### 1. 為什麼 `struct sock` 的 `skc_*` 放在結構開頭？

`__inet_lookup` 在 RX softirq 上下文被呼叫，必須低延遲。`skc_daddr`、`skc_rcv_saddr`、`skc_state`、`skc_family`、`skc_dport` 全部集中在 `sock_common` 開頭，集中在同一個 cache line。這是 Linux 2.5 → 2.6 演進中，最重要的網路最佳化之一：把「hot path」資料結構拆成 `_common` 與 `sk` 兩段，hash 查表只觸碰 64 bytes。

### 2. skb 的「共享」機制

skb 不是一份 copy 出去給應用讀。RX 進 socket 後，應用讀到 user 端是 `copy_to_user` 出去的，但若使用 `sendfile` / `splice`，skb 會直接 `skb->data` 給 `page` 引用、pipe buffer 共享，避免 copy 一次。`skb_orphan_frags` 確保 `frags[]` 中的 page 屬於使用者進程（`skb->sk`），可在 TX / 釋放時正常回收。

### 3. 為什麼 `tcp_data_queue` 區分「in-sequence」與「out-of-order」？

- 順序到達 → 立即 push 給 `sk_data_ready` callback（`sock_def_readable`） → `ep_poll_callback` 喚醒 epoll 等待者。
- 亂序 → 進入紅黑樹 `out_of_order_queue`，**不立即** `sk_data_ready`；當新資料填補 gap 時合併並一次推送，**避免高頻率 wakeup**。

這是 RACK loss detection 的基礎。

### 4. 經典悲悸：TIME_WAIT 堆積

`TCP_TIME_WAIT` 持續 60s（`TCP_TIMEWAIT_LEN`），保留 2MSL 以容忍最後一筆 ACK 丟失的 FIN 重傳。`tcp_time_wait_state_process` 對資料節流以做 perfect 匹配。若大量短連線，TIME_WAIT 數量爆炸；解法：
- `tcp_tw_reuse`：TIME_WAIT socket 可被新 SYN 復用為 source port（須 timestamp 開啟且時間遠大於 1s）。
- SO_LINGER 設 `l_linger=0` 跳過 TIME_WAIT，發 RST 直接關閉。
- `nf_conntrack_tcp_timeout_time_wait` 縮短（但違反 RFC 793）。

### 5. 經典悲悸：Socket 緩衝區積壓

應用 `read()` 慢於 NIC RX 速度時，`sk->sk_rmem_alloc` 會逼近 `sk->sk_rcvbuf`（受 `net.ipv4.tcp_rmem` 動態限制），最終 `tcp_try_rmem_schedule` 拒絕封包並丟棄；這是 **zero-window** 行為，會觸發 `tcp_should_expand_window` 與 `tcp_select_window`。在 server 端常見問題是 `skb->cloned = 1` 致 frags 不能被快速回收。

### 6. 經典悲悸：syncookies 的時機

SYN queue 滿載時，server 不再為 SYN 配置 `request_sock`，改用加密 hash 編進 SYN+ACK 的 `seq` 與 `ts`，client 回 ACK 中攜帶 hash 以驗證。**悲悸點**：syncookies 路徑 `tcp_v4_cookie_check` 看到 hash 不合會丟棄 ACK，從而 client 連線建立失敗；當 `tcp_max_syn_backlog` 設定過小或 `somaxconn` 過低時，會大量掉線。

### 7. 經典悲悸：`CLOSE_WAIT` 增長

應用層沒呼叫 `close()` 會讓 socket 卡在 `TCP_CLOSE_WAIT`。CLOSE_WAIT 表示 server 收到 client FIN 並送 ACK，但還沒送自己的 FIN。原因：應用沒關 fd（GC 沒回收、`select` 漏掉 socket、忘記 close），或 worker pool 卡死。`ss -tan state close-wait | wc -l` 是經典診斷命令。

## 實際應用推薦

### 1. 觀察 socket 統計

```sh
# 各狀態 socket 數量
ss -tan | awk 'NR>1 {print $1}' | sort | uniq -c

# 應用層讀慢 / 接收隊列堆積
ss -mem -tan 'sport = :80'   # 顯示 skmem 結構
ss -tin 'sport = :80'        # 顯示 TCP info: rtt, cwnd, retr

# 觀察 LISTEN backlog 溢位（kernel log）
dmesg | grep -E "TCP: out of memory|TCP: request_sock full"
```

### 2. 觀察 skb 與 sk 流轉

```sh
# tracepoint: kfree_skb 在哪個 sock 釋放最多
bpftrace -e 'kprobe:kfree_skb /comm != "softirq"/ { @[skb->sk->sk_state] = count(); }'

# trace skb->protocol / TCP flag 在 RX 入口
bpftrace -e 'kprobe:tcp_v4_rcv { @flags[th->ack<<3|th->psh<<2|th->rst<<1|th->syn] = count(); }'

# 看每個 socket 的 RX queue 深度
ss -mem -tan '( sport = :80 )' | awk 'NR>1 {print $5}'
```

### 3. 調校關鍵 sysctl

| 參數 | 預設 | 含義 |
|------|------|------|
| `net.ipv4.tcp_rmem` | `4096 87380 6291456` | min / default / max rcv buffer |
| `net.ipv4.tcp_wmem` | `4096 16384 4194304` | snd buffer |
| `net.ipv4.tcp_max_syn_backlog` | `1024` | SYN queue 上限 |
| `net.core.somaxconn` | `4096` | accept() backlog |
| `net.ipv4.tcp_tw_reuse` | 0 | TIME_WAIT 源埠復用 |
| `net.ipv4.tcp_slow_start_after_idle` | 1 | 閒置後 cwnd 重置 |
| `net.ipv4.tcp_fastopen` | 1 | TFO 開啟 |

### 4. 進階 trace：kprobe + eBPF

利用 `bpf_iter_sk` 與 `bpf_iter_skb` 可迭代所有 socket 與所有 skb；常見工具：
- `bpftool prog` 列出已掛載之 BPF prog。
- `libbpf-bootstrap` 範例的 `sk_collect` 自動蒐集 skb 統計。
- `funccount:tcp_v4_do_rcv` 觀察 RX hit rate。

### 5. 排查 close-wait 堆積

```sh
# 列出 close-wait 與其 process
ss -tanp state close-wait
# 對應 process 開啟的 fd
lsof -p <pid> | grep TCP
# 若多為 web server worker，懷疑 application leak — 開啟 OOM/timeout logging
```

### 6. 排查 syncookies 過早開啟

```sh
sysctl net.ipv4.tcp_max_syn_backlog      # 同步隊列上限
sysctl net.ipv4.tcp_synack_retries        # SYN+ACK 重傳次數
sysctl net.ipv4.tcp_syncookies            # 是否開啟
cat /proc/net/stat/syn_cache              # Drop 計數
```

### 7. 觀察 GRO/TSO 是否生效

```sh
ethtool -k eth0 | grep -E "tcp-segmentation-offload|generic-receive-offload"
# 觀察 softnet_stat 中的 processed/dropped/squeezed
cat /proc/net/softnet_stat
```

### 8. RDMA / kernel-bypass 取代場景

- 對 RTT < 5µs、QPS 高的應用，TCP 開銷過大。考慮：
  - **DPDK** 跳過核心 stack，user-space poll mode driver 直接 ring 3 收送。
  - **AF_XDP** 共享 NIC RX queue 給 user-space XDP map，零 copy。
  - **io_uring + TCP** 走 `io_uring_prep_send`/`recv`，但仍由核心 stack 處理。
- 對 bandwidth-bound、connection 數小但長連線場景，傳統 TCP + GRO/TSO 仍是最佳選擇。

## 結語

`sk_buff` 與 `struct sock` 是 Linux 網路堆疊的兩條骨幹：`sk_buff` 描述資料，`struct sock` 描述端點。從 `__sock_create` → `inet_create` → `tcp_prot` → `tcp_v4_init_sock` 開啟新連線，從 `tcp_sendmsg` → `tcp_transmit_skb` → `ip_queue_xmit` 把資料送出去；從 `ip_rcv` → `ip_local_deliver_finish` → `tcp_v4_rcv` → `__inet_lookup_skb` → `tcp_v4_do_rcv` → `tcp_data_queue` 把資料收進來。整個過程中，`skb_clone`、`__kfree_skb`、`skb_copy_datagram_iter`、`sk_receive_queue` 等基礎設施反覆被重用，體現了 Linux kernel 設計哲學中「資料共享、最小拷貝、cache line 對齊」的精髓。理解這些原始碼細節，等於掌握了 Linux 網路性能調校的根；任何 netstat / ss / bpftrace / perf 都只是這些結構之上的觀察儀表。
